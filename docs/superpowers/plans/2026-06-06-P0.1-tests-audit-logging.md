  status: 'success' | 'error';

### Paso 3: Crear `src/lib/audit.ts`

```typescript
import { getDb } from './firestore-admin';
import type { AuditLog } from './types';

/**
 * Registra un evento de auditoría en Firestore.
 * Se usa para detectar issues en producción sin exponerlos al usuario.
 */
export async function logAudit(log: AuditLog): Promise<void> {
  try {
    const db = getDb();
    await db.collection('audit_logs').add({
      ...log,
      timestamp: new Date().toISOString(),
    });
  } catch (err) {
    // No fallar el pedido si el logging falla
    console.error('[audit] Error logging to Firestore:', err);
  }
}
```

### Paso 2: Reescribir validaciones en `/api/order` POST

Reemplaza el bloque de validación básica con validaciones completas:

```typescript
import { logAudit } from '@/lib/audit';
import { getClientIP } from '@/lib/rate-limit';

export async function POST(request: NextRequest) {
  const start = Date.now();
  const ip = getClientIP(request);
  
  // Rate limit check (existing code)
  if (!checkRateLimit(ip, 5, 60 * 60_000)) {
    logger.warn('Rate limit excedido en /api/order', { route: '/api/order', ip });
    return NextResponse.json(
      { error: 'Demasiados pedidos. Intenta nuevamente en una hora.' },
      { status: 429 },
    );
  }

  try {
    const body = await request.json();

    // ===== VALIDACIONES MEJORADAS =====
    
    // 1. Nombre requerido y válido
    if (!body.nombre?.trim()) {
      await logAudit({
        action: 'order_validation_error',
        status: 'error',
        clientIP: ip,
        errorMessage: 'Nombre requerido',
        metadata: { nombre: body.nombre },
      });
      return NextResponse.json({ error: 'Nombre requerido' }, { status: 400 });
    }

    if (body.nombre.trim().length < 2) {
      await logAudit({
        action: 'order_validation_error',
        status: 'error',
        clientIP: ip,
        errorMessage: 'Nombre debe tener al menos 2 caracteres',
        metadata: { nombre: body.nombre },
      });
      return NextResponse.json(
        { error: 'Nombre debe tener al menos 2 caracteres' },
        { status: 400 },
      );
    }

    // 2. Dirección requerida y válida
    if (!body.direccion?.trim()) {
      await logAudit({
        action: 'order_validation_error',
        status: 'error',
        clientIP: ip,
        errorMessage: 'Dirección requerida',
      });
      return NextResponse.json({ error: 'Dirección requerida' }, { status: 400 });
    }

    // 3. Items requeridos y válidos
    if (!Array.isArray(body.items) || body.items.length === 0) {
      await logAudit({
        action: 'order_validation_error',
        status: 'error',
        clientIP: ip,
        errorMessage: 'Al menos 1 item requerido',
      });
      return NextResponse.json({ error: 'Al menos 1 item requerido' }, { status: 400 });
    }

    // 4. Validar cada item
    for (let i = 0; i < body.items.length; i++) {
      const item = body.items[i];
      if (!item.nombre || !item.cantidad || item.cantidad <= 0) {
        await logAudit({
          action: 'order_validation_error',
          status: 'error',
          clientIP: ip,
          errorMessage: `Item ${i + 1} inválido: falta nombre o cantidad`,
          metadata: { items: body.items.length },
        });
        return NextResponse.json(
          { error: `Item ${i + 1}: nombre y cantidad requeridos (cantidad > 0)` },
          { status: 400 },
        );
      }
    }

    // 5. Geo data requerida
    if (typeof body.lat !== 'number' || typeof body.lon !== 'number') {
      await logAudit({
        action: 'order_validation_error',
        status: 'error',
        clientIP: ip,
        errorMessage: 'Geolocalización inválida',
      });
      return NextResponse.json(
        { error: 'Geolocalización requerida (lat/lon)' },
        { status: 400 },
      );
    }

    // 6. Subtotal válido
    if (typeof body.subtotal !== 'number' || body.subtotal <= 0) {
      await logAudit({
        action: 'order_validation_error',
        status: 'error',
        clientIP: ip,
        errorMessage: 'Subtotal inválido',
        metadata: { subtotal: body.subtotal },
      });
      return NextResponse.json({ error: 'Subtotal debe ser > 0' }, { status: 400 });
    }

    // Si llegamos acá, todo es válido. Continuar con crearPedido()
    // (resto del código existente)

    const pedido = await crearPedido({
      nombre: body.nombre.trim(),
      direccion: body.direccion.trim(),
      telefono: encrypt(body.telefono),
      email: body.email,
      items: body.items,
      lat: body.lat,
      lon: body.lon,
      subtotal: body.subtotal,
      despacho: body.despacho || 0,
      descuento: body.descuento || 0,
      total: body.total || body.subtotal,
    });

    // Log de éxito
    await logAudit({
      action: 'order_created',
      status: 'success',
      clientIP: ip,
      pedidoId: pedido,
      metadata: {
        nombre: body.nombre,
        items: body.items.length,
        subtotal: body.subtotal,
      },
    });

    logger.info(`Pedido creado: ${pedido}`, { pedidoId: pedido, subtotal: body.subtotal });
    return NextResponse.json({ pedidoId: pedido }, { status: 201 });

  } catch (error: unknown) {
    // Log de error no previsto
    await logAudit({
      action: 'order_failed',
      status: 'error',
      clientIP: ip,
      errorMessage: error instanceof Error ? error.message : 'Unknown error',
    });

    console.error('[order POST] Error:', error);
    return NextResponse.json({ error: 'Error creando pedido' }, { status: 500 });
  }
}
```
